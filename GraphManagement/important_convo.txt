2) @tools.endpoint(...) decorator with inference

    tools_decorator.py

from __future__ import annotations
import inspect
from typing import Any, Callable, Dict, Optional, Type, get_type_hints, List
from pydantic import BaseModel
from tooling_models import (
    ToolSchema, ToolActionType, HttpMethod, AdhocToolAction, HttpToolAction
)

def _model_to_schema(model_cls: Optional[Type[BaseModel]]) -> Optional[ToolSchema]:
    if model_cls is None:
        return None
    if not (isinstance(model_cls, type) and issubclass(model_cls, BaseModel)):
        raise TypeError(f"Expected Pydantic BaseModel subclass, got {model_cls}")
    js = model_cls.model_json_schema()
    return ToolSchema(
        title=js.get("title"),
        json_schema=js,
        python_type=f"{model_cls.__module__}.{model_cls.__name__}",
    )

def _infer_io_from_signature(fn: Callable) -> Dict[str, Optional[Type[BaseModel]]]:
    """
    Infer input and output Pydantic models from function signature:
      async def fn(payload: InModel, ctx: ToolExecutionContext) -> OutModel: ...
    or:
      async def fn(payload: InModel) -> OutModel
    """
    hints = get_type_hints(fn, include_extras=True)
    sig = inspect.signature(fn)
    in_model = None
    out_model = None

    # infer input: first param annotated with BaseModel subclass
    for p in sig.parameters.values():
        annot = hints.get(p.name)
        if isinstance(annot, type) and issubclass(annot, BaseModel):
            in_model = annot
            break

    # infer output
    ret = hints.get("return")
    if isinstance(ret, type) and issubclass(ret, BaseModel):
        out_model = ret

    return {"in_model": in_model, "out_model": out_model}

def endpoint(
    *,
    name: Optional[str] = None,
    description: Optional[str] = None,  # fallback: docstring
    toolkit: Optional[str] = None,
    tags: Optional[List[str]] = None,

    # HTTP wiring (if provided -> HTTP action; else -> adhoc)
    method: Optional[HttpMethod] = None,
    path: Optional[str] = None,

    # Pydantic models (optional; inferred if omitted)
    input_model: Optional[Type[BaseModel]] = None,
    output_model: Optional[Type[BaseModel]] = None,

    # Execution hints
    idempotent: Optional[bool] = None,
    cache_ttl_seconds: Optional[int] = None,
    auth: Optional[Dict[str, Any]] = None,
    timeouts_ms: Optional[int] = None,
    rate_limit: Optional[Dict[str, Any]] = None,
):
    """
    Decorate a function to register it as a tool endpoint.
    If method+path are given -> HTTP tool; otherwise -> ad-hoc function tool.
    """
    def _decorator(fn: Callable):
        nonlocal name, description, input_model, output_model

        # infer io
        inferred = _infer_io_from_signature(fn)
        input_model = input_model or inferred["in_model"]
        output_model = output_model or inferred["out_model"]

        if description is None:
            description = inspect.getdoc(fn) or ""

        # schemas
        input_schema = _model_to_schema(input_model)
        output_schema = _model_to_schema(output_model)

        # decide action type
        if method and path:
            action = HttpToolAction(
                type="http",
                name=name or fn.__name__,
                description=description,
                tags=tags or [],
                method=method,
                path=path,
                input_schema=input_schema,
                output_schema=output_schema,
                auth=auth,
                timeouts_ms=timeouts_ms,
                rate_limit=rate_limit,
            )
        else:
            action = AdhocToolAction(
                type="adhoc",
                name=name or fn.__name__,
                description=description,
                tags=tags or [],
                input_schema=input_schema,
                output_schema=output_schema,
                python_name=fn.__name__,
                python_module=fn.__module__,
                idempotent=idempotent,
                cache_ttl_seconds=cache_ttl_seconds,
            )

        # attach metadata for discovery
        setattr(fn, "__tool_action__", action)
        setattr(fn, "__tool_meta__", {
            "tool_name": name or fn.__name__,
            "toolkit": toolkit,
            "description": description,
            "tags": tags or [],
        })
        return fn
    return _decorator

3) Discovery & export into “tool schema”

    discovery.py

from __future__ import annotations
import importlib, inspect, pkgutil
from typing import Any, Dict, Iterable, List, Optional
from pydantic import BaseModel
from tooling_models import ServiceDescriptor, Toolkit, Tool, ToolAction
from tools_decorator import endpoint

def _collect_actions_from_module(mod) -> List[ToolAction]:
    actions: List[ToolAction] = []
    for _, obj in inspect.getmembers(mod):
        if callable(obj) and hasattr(obj, "__tool_action__"):
            actions.append(getattr(obj, "__tool_action__"))
    return actions

def discover_tools_from_modules(
    modules: Iterable[str],
    *,
    default_toolkit: Optional[str] = None
) -> List[Tool]:
    tools: Dict[str, Tool] = {}
    for mod_name in modules:
        mod = importlib.import_module(mod_name)
        actions = _collect_actions_from_module(mod)
        for fn_name, obj in inspect.getmembers(mod):
            if callable(obj) and hasattr(obj, "__tool_action__"):
                action = getattr(obj, "__tool_action__")
                meta = getattr(obj, "__tool_meta__", {})
                tool_name = meta.get("tool_name") or obj.__name__
                toolkit = meta.get("toolkit") or default_toolkit or mod_name.split(".")[-1]
                description = meta.get("description") or ""
                tags = meta.get("tags") or []

                if tool_name not in tools:
                    tools[tool_name] = Tool(
                        name=tool_name,
                        description=description,
                        toolkit=toolkit,
                        version="1.0.0",
                        actions=[],
                        tags=tags,
                    )
                tools[tool_name].actions.append(action)
    return list(tools.values())

def build_service_descriptor(
    *,
    service_id: str,
    service_name: str,
    base_url: str,
    version: str,
    modules: Iterable[str],
    toolkit_name: Optional[str] = None,
    toolkit_id: Optional[str] = None,
    toolkit_desc: Optional[str] = None,
    capabilities: Optional[Dict[str, Any]] = None,
) -> ServiceDescriptor:
    tools = discover_tools_from_modules(modules, default_toolkit=toolkit_name)
    tk = Toolkit(
        id=toolkit_id or f"{service_id}:{toolkit_name or 'default'}",
        name=toolkit_name or "default",
        description=toolkit_desc,
        tools=tools,
    )
    # Bind service data onto tool entries
    for t in tk.tools:
        t.service_id = service_id
        t.service_url = base_url

    return ServiceDescriptor(
        service_id=service_id,
        service_name=service_name,
        base_url=base_url,
        version=version,
        toolkits=[tk],
        capabilities=capabilities or {},
    )

# --- OpenAI-function spec conversion (for LLM tool-calling) ---
def to_openai_tools(descriptor: ServiceDescriptor) -> List[Dict[str, Any]]:
    out: List[Dict[str, Any]] = []
    for tk in descriptor.toolkits:
        for tool in tk.tools:
            for action in tool.actions:
                if action.input_schema is None:
                    continue
                out.append({
                    "type": "function",
                    "function": {
                        "name": f"{tool.name}.{action.name}",
                        "description": action.description or tool.description or "",
                        "parameters": action.input_schema.json_schema or {"type": "object"},
                    }
                })
    return out

# Utility to serialize to dict (e.g., for a .well-known endpoint)
def to_dict(descriptor: ServiceDescriptor) -> Dict[str, Any]:
    return descriptor.model_dump()

from __future__ import annotations
from typing import List, Dict, Any, Tuple

def _flatten_tool_text(tool: Tool, action: ToolAction) -> str:
    parts = [
        tool.name, tool.description or "", tool.toolkit or "",
        action.name, action.description or "", action.type.value
    ]
    if action.input_schema and action.input_schema.json_schema:
        parts.append(str(action.input_schema.json_schema))
    if action.output_schema and action.output_schema.json_schema:
        parts.append(str(action.output_schema.json_schema))
    return "\n".join(parts)

class IndexStore:
    def __init__(self):
        self.records: List[Dict[str, Any]] = []  # {id, text, payload}

    def upsert_service(self, d: ServiceDescriptor):
        # remove old
        self.records = [r for r in self.records if r["payload"].get("service_id") != d.service_id]
        # add new
        for tk in d.toolkits:
            for tool in tk.tools:
                for action in tool.actions:
                    rid = f"{d.service_id}:{tool.name}:{action.name}"
                    self.records.append({
                        "id": rid,
                        "text": _flatten_tool_text(tool, action),
                        "payload": {
                            "service_id": d.service_id,
                            "service_url": d.base_url,
                            "tool_name": tool.name,
                            "action_name": action.name,
                            "action_type": action.type.value,
                            "toolkit": tool.toolkit,
                            "http": getattr(action, "path", None),
                            "method": getattr(action, "method", None).value if getattr(action, "method", None) else None,
                        }
                    })

    def search(self, q: str, k: int = 8) -> List[Dict[str, Any]]:
        # ultra-simple lexical scoring; replace with embeddings/ANN
        ql = q.lower()
        scored = []
        for r in self.records:
            text = r["text"].lower()
            # naive score: count overlaps
            score = sum(1 for tok in ql.split() if tok in text)
            scored.append((score, r))
        scored.sort(key=lambda t: t[0], reverse=True)
        return [dict(score=s, **r) for s, r in scored[:k]]

Graph model (nodes + rels)

    Conversation: (c:Conversation {id, title?, created_at})

    Message: (m:Message {id, role, ts, text?, tokens?, level?, embedding?})

    Block: (b:Block {id, type, mime?, text?, uri?, embedding?})

    ToolCall: (tc:ToolCall {id, tool_name, action_name, args_json, issued_at})

    ToolResult: (tr:ToolResult {id, status, output_json?, error?, latency_ms?, ended_at})

    Entity: (e:Entity {name}) (optional for mentions)

Relationships (weighted for closeness)

    (c)-[:HAS_MESSAGE]->(m)

    (m)-[:NEXT]->(m_next) (sequence)

    (m)-[:REPLY_TO]->(m_parent)

    (m)-[:HAS_BLOCK]->(b)

    (m)-[:CALLS_TOOL]->(tc)

    (tc)-[:YIELDS]->(tr)

    (m)-[:MENTIONS]->(e)

    Similarity edges (optional offline enrichment):
    (m)-[:SIMILAR {score}]->(m2) and (b)-[:SIMILAR {score}]->(b2)

Graph‑closeness retrieval (hybrid: relationship weights + similarity)

Heuristic scoring (customizable):

score = α * exp(-λ * hop_length)
      + β * (SIMILAR.score if SIMILAR exists else 0)
      + γ * temporal_decay
      + δ * tool_call overlap

Cypher sketch (structural component):

// Find K closest messages to a pivot by structure only (depth ≤ 4)
MATCH (m0:Message {id: $pivot})
MATCH p=(m0)-[r:HAS_MESSAGE|NEXT|REPLY_TO|CALLS_TOOL|YIELDS|HAS_BLOCK|MENTIONS|SIMILAR*1..4]-(m:Message)
WITH m, reduce(w=0.0, x IN relationships(p) |
  w + CASE type(x)
        WHEN 'SIMILAR' THEN 2.5
        WHEN 'REPLY_TO' THEN 1.8
        WHEN 'NEXT' THEN 1.2
        WHEN 'CALLS_TOOL' THEN 1.0
        WHEN 'YIELDS' THEN 1.0
        WHEN 'HAS_BLOCK' THEN 0.7
        WHEN 'HAS_MESSAGE' THEN 0.7
        WHEN 'MENTIONS' THEN 0.9
        ELSE 0.5 END
) AS path_weight
RETURN m.id AS message_id, max(path_weight) AS structural_score
ORDER BY structural_score DESC
LIMIT $k;

If vector search is available, blend with cosine similarity:

// Neo4j 5+ vector search example (optional)
MATCH (m0:Message {id: $pivot})
CALL db.index.vector.queryNodes('msg_emb', $k, m0.embedding) YIELD node, score
RETURN node.id AS message_id, score AS vector_score

Then fuse in app‑layer:

fused = α * normalize(structural_score) + β * normalize(vector_score)

from __future__ import annotations
import subprocess, sys, json, tempfile, textwrap
from typing import Dict, Any, Optional
from tooling_models import SnippetSpec, SnippetLanguage

class SnippetRunner:
    def __init__(self, graph_client_factory=None):
        """
        graph_client_factory: Optional callable(service_id) -> GraphClient
        used for cypher/sql snippets that need a driver.
        """
        self.graph_client_factory = graph_client_factory or (lambda _sid: None)

    def run(self, spec: SnippetSpec, input_dict: Dict[str, Any], *, service_id: Optional[str] = None) -> Dict[str, Any]:
        lang = spec.language
        if lang == SnippetLanguage.python:
            return self._run_python(spec, input_dict)
        if lang == SnippetLanguage.javascript:
            return self._run_node(spec, input_dict)
        if lang == SnippetLanguage.bash:
            return self._run_bash(spec, input_dict)
        if lang in (SnippetLanguage.sql, SnippetLanguage.cypher, SnippetLanguage.graphql):
            return self._run_graphish(spec, input_dict, service_id)
        raise ValueError(f"Unsupported snippet language: {lang}")

    def _run_python(self, spec: SnippetSpec, input_dict: Dict[str, Any]) -> Dict[str, Any]:
        # SECURITY NOTE: This uses exec for illustration only. Sandbox in production.
        src = textwrap.dedent(spec.source)
        ns: Dict[str, Any] = {}
        exec(src, ns, ns)
        entry = spec.entrypoint or "main"
        if entry not in ns or not callable(ns[entry]):
            raise ValueError(f"Python snippet missing callable '{entry}'")
        out = ns[entry](input_dict)
        if not isinstance(out, dict):
            raise TypeError("Python snippet must return dict")
        return out

    def _run_node(self, spec: SnippetSpec, input_dict: Dict[str, Any]) -> Dict[str, Any]:
        # Requires node installed in the runtime. Minimal shim; containerize later.
        with tempfile.NamedTemporaryFile("w", suffix=".mjs", delete=False) as f:
            f.write(spec.source)
            path = f.name
        entry = spec.entrypoint or "main"
        wrapper = f"import * as m from '{path}';" \
                  f"const i = JSON.parse(process.argv[2]);" \
                  f"Promise.resolve(m['{entry}'](i)).then(o=>{{" \
                  f"console.log(JSON.stringify(o||{{}})); }}).catch(e=>{{" \
                  f"console.error(e); process.exit(1); }});"
        out = subprocess.check_output(["node", "--eval", wrapper, json.dumps(input_dict)], timeout=(spec.timeout_ms or 10000)/1000.0)
        return json.loads(out.decode())

    def _run_bash(self, spec: SnippetSpec, input_dict: Dict[str, Any]) -> Dict[str, Any]:
        env = dict(**{**dict(), "INPUT_JSON": json.dumps(input_dict)})
        out = subprocess.check_output(["bash", "-c", spec.source], env=env, timeout=(spec.timeout_ms or 10000)/1000.0)
        s = out.decode().strip()
        return json.loads(s) if s else {}

    def _run_graphish(self, spec: SnippetSpec, input_dict: Dict[str, Any], service_id: Optional[str]) -> Dict[str, Any]:
        gc = self.graph_client_factory(service_id) if service_id else None
        if spec.language == SnippetLanguage.cypher:
            if not gc:
                raise RuntimeError("No GraphClient bound; cannot run Cypher snippet.")
            # Convention: provide 'params' under input; return { columns, rows, stats }
            return gc.cypher(spec.source, params=input_dict.get("params", {}), strategy=input_dict.get("strategy", "default"))
        if spec.language == SnippetLanguage.sql:
            if not gc or not hasattr(gc, "sql"):
                raise RuntimeError("No SQL client bound; cannot run SQL snippet.")
            return gc.sql(spec.source, params=input_dict.get("params", {}))
        if spec.language == SnippetLanguage.graphql:
            if not gc or not hasattr(gc, "graphql"):
                raise RuntimeError("No GraphQL client bound; cannot run GraphQL snippet.")
            return gc.graphql(spec.source, variables=input_dict.get("variables", {}))
        raise ValueError("Unsupported graph-ish snippet")

    In production you’ll swap the Python exec/Node subprocess with a sandbox or container. For now, it’s intentionally lightweight to illustrate the concept.

C) Upgrading @tools.endpoint: declare snippets and auto‑graph wiring

We keep @tools.endpoint for FastAPI tooling, but add:

from __future__ import annotations
import inspect
from typing import Any, Callable, Optional, Type, List, get_type_hints
from pydantic import BaseModel
from tooling_models import (
    ToolSchema, ToolActionType, HttpMethod,
    AdhocToolAction, HttpToolAction, SnippetToolAction, SnippetSpec
)
from graph_runtime import GraphObject, graph_automerge_inputs  # ⟵ see Section D

def endpoint(
    *,
    name: Optional[str] = None,
    description: Optional[str] = None,
    toolkit: Optional[str] = None,
    tags: Optional[List[str]] = None,

    # HTTP wiring
    method: Optional[HttpMethod] = None,
    path: Optional[str] = None,

    # Pydantic I/O (inferred if omitted)
    input_model: Optional[Type[BaseModel]] = None,
    output_model: Optional[Type[BaseModel]] = None,

    # ADHOC callable specifics
    idempotent: Optional[bool] = None,
    cache_ttl_seconds: Optional[int] = None,

    # SNIPPET specifics
    snippet: Optional[SnippetSpec] = None,
    bound_service_id: Optional[str] = None,

    # Graph auto-wiring
    graph_automerge: bool = False,
    graph_service_id: Optional[str] = None,

    # HTTP extras
    auth: Optional[dict] = None,
    timeouts_ms: Optional[int] = None,
    rate_limit: Optional[dict] = None,
):
    """
    - If snippet is provided -> registers a SnippetToolAction.
    - Else if method+path -> HttpToolAction.
    - Else -> Adhoc callable tool.
    - If graph_automerge=True, any GraphObject instances found in 'input_model' will be
      MERGE'd into the remote graph microservice before your function runs.
    """
    def _decorator(fn: Optional[Callable] = None):
        nonlocal name, description, input_model, output_model
        def _model_to_schema(model_cls: Optional[Type[BaseModel]]):
            if model_cls is None: return None
            js = model_cls.model_json_schema()
            return ToolSchema(title=js.get("title"), json_schema=js, python_type=f"{model_cls.__module__}.{model_cls.__name__}")

        def _infer_io(_fn: Callable):
            hints = get_type_hints(_fn, include_extras=True)
            in_model = None; out_model = None
            for p in inspect.signature(_fn).parameters.values():
                annot = hints.get(p.name)
                if isinstance(annot, type) and issubclass(annot, BaseModel):
                    in_model = annot; break
            ret = hints.get("return")
            if isinstance(ret, type) and issubclass(ret, BaseModel):
                out_model = ret
            return in_model, out_model

        def _attach_action(_fn: Optional[Callable]):
            nonlocal input_model, output_model
            if description is None:
                desc = inspect.getdoc(_fn) if _fn else ""
            else:
                desc = description

            if _fn and (input_model is None or output_model is None):
                inf_in, inf_out = _infer_io(_fn)
                input_model = input_model or inf_in
                output_model = output_model or inf_out

            inp_schema = _model_to_schema(input_model)
            out_schema = _model_to_schema(output_model)

            if snippet is not None:
                action = SnippetToolAction(
                    type="snippet",
                    name=name or (_fn.__name__ if _fn else "snippet"),
                    description=desc or "",
                    tags=tags or [],
                    input_schema=inp_schema,
                    output_schema=out_schema,
                    snippet=snippet,
                    idempotent=idempotent,
                    cache_ttl_seconds=cache_ttl_seconds,
                    bound_service_id=bound_service_id,
                )
            elif method and path:
                action = HttpToolAction(
                    type="http",
                    name=name or (_fn.__name__ if _fn else "http_action"),
                    description=desc or "",
                    tags=tags or [],
                    method=method, path=path,
                    input_schema=inp_schema, output_schema=out_schema,
                    auth=auth, timeouts_ms=timeouts_ms, rate_limit=rate_limit
                )
            else:
                action = AdhocToolAction(
                    type="adhoc",
                    name=name or _fn.__name__,
                    description=desc or "",
                    tags=tags or [],
                    input_schema=inp_schema, output_schema=out_schema,
                    python_name=_fn.__name__, python_module=_fn.__module__,
                    idempotent=idempotent, cache_ttl_seconds=cache_ttl_seconds,
                )

            meta = {
                "tool_name": name or (_fn.__name__ if _fn else "tool"),
                "toolkit": toolkit,
                "description": desc or "",
                "tags": tags or [],
                "graph_automerge": graph_automerge,
                "graph_service_id": graph_service_id,
            }

            # Attach to fn (or a dummy callable for pure snippets)
            target = _fn or (lambda *a, **k: None)
            setattr(target, "__tool_action__", action)
            setattr(target, "__tool_meta__", meta)
            return target

        if fn is not None:
            wrapped = fn
            if graph_automerge and input_model is not None:
                # auto-merge GraphObjects appearing in input before calling fn
                async def wrapper(payload, *args, **kwargs):
                    payload = graph_automerge_inputs(payload, service_id=graph_service_id)
                    return await fn(payload, *args, **kwargs)
                wrapped = _attach_action(wrapper)
            else:
                wrapped = _attach_action(fn)
            return wrapped

        # decorator used as @endpoint(...)(fn)
        def inner(_fn: Callable):
            return _attach_action(_fn)
        return inner
    return _decorator

from __future__ import annotations
from typing import Any, Dict, List, Optional, Type, Iterable, Union
from pydantic import BaseModel, Field
import httpx
from dataclasses import dataclass
from copy import deepcopy

# --- Client ---------------------------------------------------------

class GraphClient:
    def __init__(self, base_url: str, api_key: Optional[str] = None, timeout: float = 15.0):
        self.base_url = base_url.rstrip("/")
        self.headers = {"Authorization": f"Bearer {api_key}"} if api_key else {}
        self.timeout = timeout

    def cypher(self, query: str, params: Dict[str, Any] = None, strategy: str = "default") -> Dict[str, Any]:
        body = {"query": query, "strategy": strategy, "params": params or {}}
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/cypher", json=body)
            r.raise_for_status()
            return r.json()

    def merge_node(self, label: str, key: Dict[str, Any], props: Dict[str, Any] = None) -> Dict[str, Any]:
        payload = {"label": label, "key": key, "props": props or {}}
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/nodes/merge", json=payload)
            r.raise_for_status()
            return r.json()

    def merge_rel(self, out_label: str, out_key: Dict[str, Any], rel_type: str, in_label: str, in_key: Dict[str, Any], props: Dict[str, Any] = None) -> Dict[str, Any]:
        payload = {"out": {"label": out_label, "key": out_key},
                   "rel": {"type": rel_type, "props": props or {}},
                   "in":  {"label": in_label,  "key": in_key}}
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/rels/merge", json=payload)
            r.raise_for_status()
            return r.json()

# Simple registry so decorators can find the client by service_id.
_GRAPH_CLIENTS: Dict[str, GraphClient] = {}
def register_graph_client(service_id: str, base_url: str, api_key: Optional[str] = None):
    _GRAPH_CLIENTS[service_id] = GraphClient(base_url, api_key=api_key)

def get_graph_client(service_id: str) -> GraphClient:
    if service_id not in _GRAPH_CLIENTS:
        raise RuntimeError(f"No GraphClient registered for service_id={service_id}")
    return _GRAPH_CLIENTS[service_id]

# --- Graph Objects --------------------------------------------------

@dataclass
class GraphMeta:
    label: str
    key_fields: List[str]
    service_id: str
    persist_on_init: bool = False

class GraphObject(BaseModel):
    """Base class for remote graph-backed objects."""
    __graph__: GraphMeta

    def _graph_identity(self) -> Dict[str, Any]:
        return {k: getattr(self, k) for k in self.__graph__.key_fields}

    def _graph_props(self) -> Dict[str, Any]:
        # props = all model fields minus key fields
        return {k: getattr(self, k) for k in self.model_fields if k not in self.__graph__.key_fields}

    def merge(self):
        g = get_graph_client(self.__graph__.service_id)
        return g.merge_node(self.__graph__.label, self._graph_identity(), self._graph_props())

    def relate_to(self, other: "GraphObject", rel_type: str, props: Dict[str, Any] = None):
        g = get_graph_client(self.__graph__.service_id)
        return g.merge_rel(self.__graph__.label, self._graph_identity(), rel_type, other.__graph__.label, other._graph_identity(), props or {})

def object(label: str, key: Union[str, List[str]], service_id: str, persist_on_init: bool = False):
    """
    Decorate a Pydantic model class -> remote GraphObject.
    Persist immediately on construction if persist_on_init=True.
    """
    def _decorator(cls: Type[BaseModel]):
        key_fields = [key] if isinstance(key, str) else list(key)
        # Create dynamic subclass mixing GraphObject
        name = f"{cls.__name__}Graph"
        attrs = dict(cls.__dict__)
        # Build a new Pydantic model subclassing both GraphObject and original
        NewCls = type(name, (GraphObject, cls), {})
        NewCls.__graph__ = GraphMeta(label=label, key_fields=key_fields, service_id=service_id, persist_on_init=persist_on_init)

        # Hook model_post_init to auto-merge if requested
        orig_post = getattr(NewCls, "model_post_init", None)
        def model_post_init(self, __ctx):
            if orig_post: orig_post(self, __ctx)
            if NewCls.__graph__.persist_on_init:
                self.merge()
        NewCls.model_post_init = model_post_init

        return NewCls
    return _decorator

# --- Auto-merge traversal ------------------------------------------

def graph_automerge_inputs(payload: Any, *, service_id: Optional[str]) -> Any:
    """
    Walks a Pydantic input model instance; for any GraphObject (or list of them), MERGE into remote graph.
    Returns a deep-copied payload (non-destructive) with the same structure.
    """
    def _visit(x: Any):
        if isinstance(x, GraphObject):
            # Ensure service alignment
            if service_id and x.__graph__.service_id != service_id:
                raise RuntimeError(f"Graph service mismatch: decorator={service_id}, obj={x.__graph__.service_id}")
            x.merge()
            return x
        if isinstance(x, list):
            return [_visit(i) for i in x]
        if isinstance(x, dict):
            return {k: _visit(v) for k, v in x.items()}
        return x

    if hasattr(payload, "model_dump"):
        dumped = payload.model_dump()
        _visit(payload)  # merges side effect
        return payload  # keep Pydantic instance for handler
    if isinstance(payload, dict):
        _visit(payload)
        return payload
    return payload

    Uniform methods: .merge() and .relate_to(...).

    graph_automerge_inputs(...) lets @tools.endpoint auto‑persist GraphObjects that appear in the input payload.

from tooling_models import SnippetToolAction, AdhocToolAction, HttpToolAction, ToolAction

import httpx, importlib

def execute(action: ToolAction, args: Dict[str, Any], meta: Dict[str, Any]) -> Dict[str, Any]:
    if isinstance(action, SnippetToolAction):
        runner = SnippetRunner(graph_client_factory=lambda sid: get_graph_client(sid))

from __future__ import annotations
from typing import Any, Callable, Dict, List, Literal, Optional, Sequence, Union
from pydantic import BaseModel, Field
from enum import Enum

# ========= Embeddings =========

class SimilarityFn(str, Enum):
    cosine = "cosine"
    dot = "dot"
    euclidean = "euclidean"

class EmbedWhen(str, Enum):
    create = "create"
    update = "update"

class EmbeddingSource(BaseModel):
    """
    Where to pull text from to embed.
    Either:
      - field: str                     (single field name)
      - fields: list[str]              (concatenate)
      - fn: Callable[[Any], str]       (custom compute)
    """
    field: Optional[str] = None
    fields: Optional[List[str]] = None
    # NOTE: callables can't be serialized to JSON; OK for in-process config.
    fn: Optional[Callable[[Any], str]] = None
    sep: str = "\n"

    def extract(self, obj: Any) -> str:
        if self.fn:
            return self.fn(obj)
        if self.field:
            return getattr(obj, self.field, "") or ""
        if self.fields:
            parts = [str(getattr(obj, f, "") or "") for f in self.fields]
            return self.sep.join([p for p in parts if p])
        return ""

class EmbeddingSpec(BaseModel):
    """
    Defines a named embedding space for an object.
    """
    name: str
    provider: str  # e.g. "openai"|"text-embeddings-local"|"fastembed"
    model: str     # e.g. "text-embedding-3-large"
    dim: int
    similarity: SimilarityFn = SimilarityFn.cosine
    source: EmbeddingSource
    when: List[EmbedWhen] = Field(default_factory=lambda: [EmbedWhen.create, EmbedWhen.update])
    store_property: str | None = None   # property name to store vector; else driver-side vector column
    index_name: str | None = None       # vector index identifier in graph
    autolink: Optional["AutoLinkSimilar"] = None

class AutoLinkSimilar(BaseModel):
    """
    Autogenerate :SIMILAR edges using this embedding space.
    """
    k: int = 20
    threshold: float = 0.78
    weight: float = 2.5
    # optional scope filtering
    label_scope: Optional[str] = None   # restrict to same label or a specific label
    rel_type: str = "SIMILAR"
    bidirectional: bool = True

EmbeddingSpec.model_rebuild()

    Declarative relation wiring.
    - static endpoint: to_label + to_key mapping (dict of field → prop)
    - dynamic endpoint: from object field(s) that contain GraphObjects or scalars (resolved to nodes)
    """
    name: str                  # rel type e.g. "BELONGS_TO", "MENTIONS"
    direction: Direction = Direction.OUT
    cardinality: Cardinality = Cardinality.MANY_TO_MANY
    to_label: Optional[str] = None
    to_key: Optional[List[str]] = None
    props: Dict[str, Any] = Field(default_factory=dict)
    schema: Optional[EdgeSchema] = None

    # Dynamic sources (one of these can be set)
    from_field: Optional[str] = None        # field that yields GraphObject(s) or scalars
    resolve_to_label: Optional[str] = None  # if scalars, create/merge nodes of this label
    resolve_key: Optional[List[str]] = None # fields used as key on resolved nodes

    # Hooks (optional)
    on_create: Optional[Callable[[Any, Any], None]] = None  # (self, graph_client)
    on_update: Optional[Callable[[Any, Any], None]] = None

# ----- Graph client extensions -----

class GraphClient:
    def __init__(self, base_url: str, api_key: Optional[str] = None, timeout: float = 20.0):
        self.base_url = base_url.rstrip("/")
        self.headers = {"Authorization": f"Bearer {api_key}"} if api_key else {}
        self.timeout = timeout

    # Basic upsert
    def merge_node(self, label: str, key: Dict[str, Any], props: Dict[str, Any] = None, vectors: Dict[str, List[float]] = None) -> Dict[str, Any]:
        body = {"label": label, "key": key, "props": props or {}, "vectors": vectors or {}}
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/nodes/merge", json=body)
            r.raise_for_status()
            return r.json()

    def merge_rel(self, out_label: str, out_key: Dict[str, Any], rel_type: str,
                  in_label: str, in_key: Dict[str, Any], props: Dict[str, Any] = None) -> Dict[str, Any]:
        body = {"out": {"label": out_label, "key": out_key},
                "rel": {"type": rel_type, "props": props or {}},
                "in":  {"label": in_label,  "key": in_key}}
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/rels/merge", json=body)
            r.raise_for_status()
            return r.json()

    # Vector search for autolinks
    def vector_search(self, label: str, space: str, vector: List[float], k: int, threshold: float,
                      similarity: str = "cosine") -> List[Dict[str, Any]]:
        body = {
            "label": label, "space": space, "vector": vector,
            "k": k, "threshold": threshold, "similarity": similarity
        }
        with httpx.Client(timeout=self.timeout, headers=self.headers) as c:
            r = c.post(f"{self.base_url}/vectors/search", json=body)
            r.raise_for_status()
            return r.json().get("results", [])

# Central registry (reuse your existing one if present)

from __future__ import annotations
from typing import Any, Dict, List, Optional, Type, Union
from pydantic import BaseModel
from dataclasses import dataclass
from graph_overlay_specs import (
    EmbeddingSpec, RelationSpec, FieldSpec, CalculatedFieldSpec, MemoryPolicies,
    AutoLinkSimilar, EmbedWhen, CalcWhen, Direction
)
from graph_overlay_runtime import get_graph_client, get_embedder

@dataclass
class OverlayMeta:
    label: str
    key_fields: List[str]
    service_id: str
    persist_on_init: bool
    embeddings: List[EmbeddingSpec]
    relations: List[RelationSpec]
    fields: List[FieldSpec]
    calc_fields: List[CalculatedFieldSpec]
    policies: MemoryPolicies

class GraphObject(BaseModel):
    __overlay__: OverlayMeta

    # Identity/props helpers
    def _identity(self) -> Dict[str, Any]:
        return {k: getattr(self, k) for k in self.__overlay__.key_fields}

    def _all_props(self) -> Dict[str, Any]:
        return {k: getattr(self, k) for k in self.model_fields}

    # ------- Calculated fields -------
    def _apply_calculated(self, when: CalcWhen):
        for cf in self.__overlay__.calc_fields:
            if when in cf.when:
                val = cf.fn(self)
                if cf.store:
                    setattr(self, cf.name, val)

    # ------- Embeddings -------
    def _compute_embeddings(self, when: EmbedWhen) -> Dict[str, List[float]]:
        vectors: Dict[str, List[float]] = {}
        for es in self.__overlay__.embeddings:
            if when not in es.when:
                continue
            text = es.source.extract(self) or ""
            if not text.strip():
                continue
            embedder = get_embedder(es.provider)
            vec = embedder(text, model=es.model, dim=es.dim)  # must return list[float]
            if not isinstance(vec, list) or len(vec) != es.dim:
                raise ValueError(f"Embedder returned invalid vector for space '{es.name}'")
            vectors[es.name] = vec
        return vectors

    # ------- Relations -------
    def _materialize_relations(self, gc, vectors: Dict[str, List[float]] | None = None):
        """
        Build declared relations (MENTIONS, BELONGS_TO, etc.) and autolink SIMILAR if configured.
        """
        # 1) Declarative relations
        for rel in self.__overlay__.relations:
            # Hooks first (allow custom side-effects)
            if hasattr(rel, "on_create") and callable(rel.on_create):
                try:
                    rel.on_create(self, gc)
                except Exception:
                    pass

            # Dynamic from_field
            if rel.from_field:
                src_val = getattr(self, rel.from_field, None)
                if src_val is None:
                    continue

                items = src_val if isinstance(src_val, list) else [src_val]
                for item in items:
                    # Case A: item is another GraphObject -> relate directly
                    if hasattr(item, "__overlay__"):
                        out_label, out_key = self.__overlay__.label, self._identity()
                        in_label, in_key = item.__overlay__.label, item._identity()
                        if rel.direction == Direction.OUT:
                            gc.merge_rel(out_label, out_key, rel.name, in_label, in_key, rel.props)
                        elif rel.direction == Direction.IN:
                            gc.merge_rel(in_label, in_key, rel.name, out_label, out_key, rel.props)
                        else:  # BOTH
                            gc.merge_rel(out_label, out_key, rel.name, in_label, in_key, rel.props)
                            gc.merge_rel(in_label, in_key, rel.name, out_label, out_key, rel.props)
                        continue

                    # Case B: scalar -> resolve to node using resolve_* hints
                    if rel.resolve_to_label and rel.resolve_key:
                        in_label = rel.resolve_to_label
                        # scalar maps to the first key field
                        in_key = {rel.resolve_key[0]: item}
                        out_label, out_key = self.__overlay__.label, self._identity()
                        if rel.direction == Direction.OUT:
                            gc.merge_rel(out_label, out_key, rel.name, in_label, in_key, rel.props)
                        elif rel.direction == Direction.IN:
                            gc.merge_rel(in_label, in_key, rel.name, out_label, out_key, rel.props)
                        else:
                            gc.merge_rel(out_label, out_key, rel.name, in_label, in_key, rel.props)
                            gc.merge_rel(in_label, in_key, rel.name, out_label, out_key, rel.props)

            # Static endpoint if provided (e.g., BELONGS_TO Conversation)
            elif rel.to_label and rel.to_key:
                out_label, out_key = self.__overlay__.label, self._identity()
                in_label, in_key = rel.to_label, {k: getattr(self, k) for k in rel.to_key}
                gc.merge_rel(out_label, out_key, rel.name, in_label, in_key, rel.props)

        # 2) Auto link SIMILAR if vectors provided
        for es in self.__overlay__.embeddings:
            if not es.autolink:
                continue
            if not vectors or es.name not in vectors:
                continue
            al = es.autolink
            search_label = al.label_scope or self.__overlay__.label
            hits = gc.vector_search(
                label=search_label,
                space=es.name,
                vector=vectors[es.name],
                k=al.k,
                threshold=al.threshold,
                similarity=es.similarity.value if hasattr(es.similarity, "value") else str(es.similarity),
            )
            me_label, me_key = self.__overlay__.label, self._identity()
            for h in hits:
                if h.get("key") == me_key:  # skip self
                    continue
                other_label = h.get("label", search_label)
                other_key = h.get("key", {})
                gc.merge_rel(me_label, me_key, al.rel_type, other_label, other_key, {"weight": al.weight})
                if al.bidirectional:
                    gc.merge_rel(other_label, other_key, al.rel_type, me_label, me_key, {"weight": al.weight})

    # ------- Persist -------
    def merge(self, *, is_update: bool = False):
        gc = get_graph_client(self.__overlay__.service_id)

        # 1) calculated
        self._apply_calculated(CalcWhen.update if is_update else CalcWhen.create)

        # 2) embeddings
        vectors = self._compute_embeddings(EmbedWhen.update if is_update else EmbedWhen.create)

        # 3) filter/stage props (respect FieldSpec.store=False)
        props = self._all_props()
        if self.__overlay__.fields:
            keep = {f.name for f in self.__overlay__.fields if f.store}
            props = {k: v for k, v in props.items() if (not self.__overlay__.fields or k in keep or k in self.__overlay__.key_fields)}

        # 4) merge node + vectors
        gc.merge_node(self.__overlay__.label, self._identity(), props, vectors=vectors)

        # 5) relations (including similarity autolinks + mentions)
        self._materialize_relations(gc, vectors=vectors)

def object(
    *,
    label: str,
    key: Union[str, List[str]],
    service_id: str,
    persist_on_init: bool = False,

    embeddings: Optional[List[EmbeddingSpec]] = None,
    relations: Optional[List[RelationSpec]] = None,
    fields: Optional[List[FieldSpec]] = None,
    calc_fields: Optional[List[CalculatedFieldSpec]] = None,
    policies: Optional[MemoryPolicies] = None,
):
    """
    Upgrade a Pydantic model class to a graph-backed memory object with embeddings,
    relationships, structured & calculated fields, and overlay policies.
    """
    def _decorator(cls: Type[BaseModel]):
        key_fields = [key] if isinstance(key, str) else list(key)

        name = f"{cls.__name__}Overlay"
        NewCls = type(name, (GraphObject, cls), {})
        NewCls.__overlay__ = OverlayMeta(
            label=label,
            key_fields=key_fields,
            service_id=service_id,
            persist_on_init=persist_on_init,
            embeddings=embeddings or [],
            relations=relations or [],
            fields=fields or [],
            calc_fields=calc_fields or [],
            policies=policies or MemoryPolicies(),
        )

        # Auto-persist on construction if requested
        orig_post = getattr(NewCls, "model_post_init", None)
        def model_post_init(self, __ctx):
            if orig_post: orig_post(self, __ctx)
            if NewCls.__overlay__.persist_on_init:
                self.merge(is_update=False)
        NewCls.model_post_init = model_post_init
        return NewCls
    return _decorator

# memory_calcs.py
import math, time
from typing import Any

def calc_tokens(field: str):
    def _fn(obj: Any) -> int:
        txt = getattr(obj, field, "") or ""
        # naive token count (swap with tokenizer later)
        return len(txt.strip().split())
    return _fn

def calc_salience(score_field: str = "score",
                  created_field: str = "created_ts",
                  updated_field: str = "updated_ts",
                  alpha: float = 0.6, beta: float = 0.3, gamma: float = 0.1,
                  half_life_seconds: float = 7*24*3600.0):  # 7 days
    """
    Exponential recency decay + frequency proxy (updates) + explicit score.
    """
    def _fn(obj: Any) -> float:
        now = time.time()
        created = getattr(obj, created_field, None) or now
        updated = getattr(obj, updated_field, None) or created
        score = float(getattr(obj, score_field, 0.0) or 0.0)

        age = max(0.0, now - created)
        recency = math.exp(-math.log(2.0) * age / half_life_seconds)

        # frequency proxy = number of updates if available
        freq = 1.0 + max(0.0, (updated - created)) / (24*3600.0)  # 1 + days since create

        s = alpha*recency + beta*math.tanh(freq/10.0) + gamma*math.tanh(score/10.0)
        return float(round(s, 6))
    return _fn

7) Notes on structured fields & policies

    FieldSpec is metadata. Use it to:

        Materialize indexes/constraints during a migration step (create UNIQUE on keys, FULLTEXT on title/text, etc.).

        Mark PII for redaction on export.

        Mark encryption if you layer KMS at the microservice boundary.

    MemoryPolicies.versioned=True → on merge, you can:

        copy old props to a :MemoryItemVersion node + (:MemoryItem)-[:HAS_VERSION]->(:MemoryItemVersion)

        or stamp version number and create (:MemoryItem)-[:PREV]->(:MemoryItem) history chain.

    ttl_seconds in MemoryPolicies (or FieldSpec.ttl_seconds) → schedule GC jobs to expire stale memories or specific fields.

    Extends the executor to actually run/await/stream these shapes and persist them as graph objects (:Job, :Stream, :Chunk) for the conversation graph/memory overlay.

I’ve kept names consistent with what we built earlier so you can paste these in with minimal friction.
0) High‑level: how it fits together

    BasedModel ⇢ GraphObject with nicer API (save(), link(), unlink(), neighbors()), compatible with @graph.object(...).

    Type descriptors ⇢ TypeSemantics that annotate output schema with:

        kind: "immediate" | "awaitable" | "stream" | "job" | "union" | "recursive"

        type_expr: pretty python typing string (e.g., AsyncIterator[ChunkOut])

        item_schema for streams, result_schema for awaitables/jobs, oneOf + discriminator for unions.

    Graph representation for async/streaming:

        (:ToolCall)-[:SPAWNS]->(:Job) with status machine and final (:ToolResult).

        (:ToolCall)-[:STREAMS]->(:Stream) with (:Chunk {seq, ts, data}).

    Discovery exports output_semantics so your registry and planners know how to use/compose them.

1) BasedModel: a graph‑ORM flavored base

from __future__ import annotations
from typing import Any, Dict, List, Optional
from pydantic import BaseModel
from graph_overlay import GraphObject  # from your previous overlay
from graph_overlay_runtime import get_graph_client

class BasedModel(GraphObject):
    """
    A nicer façade over GraphObject with ORM-like ergonomics.
    Works seamlessly with @graph.object(...) since GraphObject is the MRO base.
    """

    # Persistence
    def save(self, *, update: bool = False):
        return self.merge(is_update=update)

    # Relations
    def link(self, rel_type: str, other: "BasedModel", props: Optional[Dict[str, Any]] = None):
        return self.relate_to(other, rel_type, props or {})

    def unlink(self, rel_type: str, other: "BasedModel"):
        gc = get_graph_client(self.__overlay__.service_id)
        return gc.merge_rel(  # simple delete via merge+delete fallback on backend if supported
            self.__overlay__.label, self._identity(), f"__UNLINK__:{rel_type}",
            other.__overlay__.label, other._identity(), {}
        )

    # Traversals (thin convenience; you likely have richer server APIs)
    def neighbors(self, rel_type: Optional[str] = None, direction: str = "OUT", limit: int = 50):
        gc = get_graph_client(self.__overlay__.service_id)
        # A generic traversal endpoint would be ideal; here’s a stub contract suggestion:
        # POST /traverse {start: {label,key}, rel_type, direction, limit}
        # return list of nodes as {label, key, props}
        try:
            return gc.traverse(
                start={"label": self.__overlay__.label, "key": self._identity()},
                rel_type=rel_type, direction=direction, limit=limit
            )
        except AttributeError:
            return []  # backend not implemented yet

2) Async/advanced type semantics: describing “weird” outputs

We’ll encode typing details alongside JSON Schema so planners and UIs can handle async, streams, unions, and recursion.
2.1 Type descriptor models

    async_type_spec.py

from __future__ import annotations
import inspect, asyncio, typing as t
from typing import Any, Dict, Optional, List, get_origin, get_args
from pydantic import BaseModel, TypeAdapter

# ---- Public export put into actions as "output_semantics" ----
class TypeSemantics(BaseModel):
    kind: t.Literal["immediate","awaitable","stream","job","union","recursive"] = "immediate"
    type_expr: str
    # For awaitable/job
    result_schema: Optional[Dict[str, Any]] = None
    # For streams
    item_schema: Optional[Dict[str, Any]] = None
    transport: Optional[t.Literal["ndjson","sse","websocket","http-chunked"]] = None
    # For unions
    one_of: Optional[List[Dict[str, Any]]] = None
    discriminator: Optional[str] = None
    # For recursive (best-effort; refs by title)
    refs: Optional[Dict[str, Dict[str, Any]]] = None
    # Raw JSON schema for the top-level when it’s a plain type
    json_schema: Optional[Dict[str, Any]] = None
    # Python-specific typing details
    x_typing: Optional[Dict[str, Any]] = None

# ---- internal helpers ----
ASYNC_STREAM_ORIGINS = {
    t.AsyncIterator, t.AsyncIterable, t.AsyncGenerator
}
AWAITABLE_ORIGINS = {
    t.Awaitable, t.Coroutine, asyncio.Future, asyncio.Task
}

def _type_expr(tp: Any) -> str:
    try:
        return str(tp).replace("typing.", "")
    except Exception:
        return repr(tp)

def _json_schema(tp: Any) -> Optional[Dict[str, Any]]:
    try:
        return TypeAdapter(tp).json_schema()
    except Exception:
        return None

def _is_union(tp: Any) -> bool:
    return get_origin(tp) is t.Union or getattr(tp, "__origin__", None) is t.Union

def _is_annotated(tp: Any) -> bool:
    return get_origin(tp) is t.Annotated

def _unwrap_annotated(tp: Any) -> Any:
    return get_args(tp)[0] if _is_annotated(tp) else tp

def describe_type(tp: Any) -> TypeSemantics:
    """Produce a rich description of Python typing -> JSON Schema + async semantics."""
    original = tp
    tp = _unwrap_annotated(tp)
    origin = get_origin(tp) or tp
    args = get_args(tp)

    # Streams
    if origin in ASYNC_STREAM_ORIGINS:
        item = args[0] if args else Any
        return TypeSemantics(
            kind="stream",
            type_expr=_type_expr(original),
            item_schema=_json_schema(item) or {"type": "object"},
            transport="ndjson",
            x_typing={"origin": str(origin), "args": [_type_expr(a) for a in args]},
        )

    # Awaitables (Coroutine[..., T], Awaitable[T], Future[T], Task[T])
    if origin in AWAITABLE_ORIGINS:
        result = args[-1] if args else Any
        return TypeSemantics(
            kind="awaitable",
            type_expr=_type_expr(original),
            result_schema=_json_schema(result) or {"type": "object"},
            x_typing={"origin": str(origin), "args": [_type_expr(a) for a in args]},
        )

    # Union / Optional
    if _is_union(tp):
        options = []
        for opt in get_args(tp):
            options.append(_json_schema(opt) or {"type": "object"})
        # Best-effort discriminator if any option is a Pydantic model with a 'type' field
        return TypeSemantics(
            kind="union",
            type_expr=_type_expr(original),
            one_of=options,
            discriminator=None,
            x_typing={"union": [_type_expr(o) for o in get_args(tp)]},
        )

    # Recursion: Pydantic handles $ref. If schema has $defs with self-refs, mark as recursive.
    js = _json_schema(tp)
    is_recursive = bool(js and ("$defs" in js or "definitions" in js))
    if is_recursive:
        refs = js.get("$defs") or js.get("definitions")
        return TypeSemantics(
            kind="recursive",
            type_expr=_type_expr(original),
            json_schema=js,
            refs=refs,
        )

    # Plain immediate type
    return TypeSemantics(
        kind="immediate",
        type_expr=_type_expr(original),
        json_schema=js or {"type": "object"},
    )

6) Executor that awaits, streams, and persists to graph

We’ll add an execution layer that:

    awaits Awaitable[T]/Coroutine[..., T]/Future[T]/Task[T]

    streams AsyncIterator[T] into graph chunks and optionally returns a StreamRef

    immediate returns are unchanged

    Optionally promotes long‑running awaitables to jobs (if you want non-blocking semantics) by returning a JobRef

    graph_job_stream.py

from __future__ import annotations
import asyncio, time
from typing import Any, Dict, Optional, AsyncIterator, Union, List
from pydantic import BaseModel
from graph_overlay_runtime import get_graph_client

class JobRef(BaseModel):
    job_id: str
    status: str
    status_endpoint: Optional[str] = None
    stream_endpoint: Optional[str] = None
    result_schema: Optional[Dict[str, Any]] = None

class StreamRef(BaseModel):
    stream_id: str
    transport: str = "ndjson"
    item_schema: Optional[Dict[str, Any]] = None

class GraphRunRecorder:
    """
    Persist jobs/streams to the graph service. Minimal contract; backend can be MERGE-based.
    """
    def __init__(self, service_id: str):
        self.service_id = service_id
        self.gc = get_graph_client(service_id)

    # Jobs
    def job_start(self, tool_call: Dict[str, Any], meta: Dict[str, Any]) -> JobRef:
        job_id = tool_call.get("id")
        props = {"status": "running", "ts": time.time(), "tool": tool_call.get("tool_name")}
        self.gc.merge_node("Job", {"id": job_id}, props)
        self.gc.merge_rel("ToolCall", {"id": tool_call["id"]}, "SPAWNS", "Job", {"id": job_id}, {})
        return JobRef(job_id=job_id, status="running")

    def job_finish(self, job_id: str, ok: bool, result: Optional[Dict[str, Any]] = None, error: Optional[str] = None):
        props = {"status": "done" if ok else "error", "ended_ts": time.time()}
        self.gc.merge_node("Job", {"id": job_id}, props)
        if result is not None:
            # attach a ToolResult node and link
            self.gc.merge_node("ToolResult", {"id": job_id}, {"status": "ok", "output": result})
            self.gc.merge_rel("Job", {"id": job_id}, "YIELDS", "ToolResult", {"id": job_id}, {})
        if error:
            self.gc.merge_node("ToolResult", {"id": job_id}, {"status": "error", "error": error})
            self.gc.merge_rel("Job", {"id": job_id}, "YIELDS", "ToolResult", {"id": job_id}, {})

    # Streams
    def stream_start(self, tool_call: Dict[str, Any], item_schema: Optional[Dict[str, Any]] = None, transport: str = "ndjson") -> StreamRef:
        sid = tool_call.get("id") + ":stream"
        self.gc.merge_node("Stream", {"id": sid}, {"transport": transport})
        self.gc.merge_rel("ToolCall", {"id": tool_call["id"]}, "STREAMS", "Stream", {"id": sid}, {})
        return StreamRef(stream_id=sid, transport=transport, item_schema=item_schema)

    def stream_append(self, stream_id: str, seq: int, data: Dict[str, Any]):
        # Represent each chunk as a Block or Chunk node
        self.gc.merge_node("Chunk", {"id": f"{stream_id}:{seq}"}, {"seq": seq, "ts": time.time(), "data": data})
        self.gc.merge_rel("Stream", {"id": stream_id}, "HAS_CHUNK", "Chunk", {"id": f"{stream_id}:{seq}"}, {})

from __future__ import annotations
import inspect, asyncio
from typing import Any, Dict, AsyncIterator, get_origin, get_args
from tooling_models import ToolAction, SnippetToolAction, AdhocToolAction, HttpToolAction
from async_type_spec import describe_type
from graph_job_stream import GraphRunRecorder
# ... reuse your previous SnippetRunner/http executor imports ...

async def _maybe_await(x: Any) -> Any:
    if inspect.isawaitable(x):
        return await x
    return x

async def _maybe_stream(x: Any) -> AsyncIterator[Any] | None:
    # AsyncIterator/AsyncGenerator
    origin = get_origin(type(x)) or type(x)
    if inspect.isasyncgen(x) or origin.__name__ in ("async_generator",):
        return x
    # Best-effort duck-typing: has __aiter__
    if hasattr(x, "__aiter__"):
        return x
    return None

async def execute_async(action: ToolAction, args: Dict[str, Any], meta: Dict[str, Any], tool_call_ctx: Dict[str, Any]) -> Dict[str, Any]:
    """
    Executes a tool action with async/stream semantics and persists Job/Stream to graph.
    Returns a payload appropriate for your gateway (immediate result / JobRef / StreamRef).
    """
    recorder = GraphRunRecorder(service_id=meta.get("graph_service_id") or meta.get("service_id") or meta.get("service"))
    # 1) Run the underlying action using your existing machinery
    # (We re-use your earlier executor; here assume we have a sync 'execute' you can call.)
    from executor import execute as execute_sync  # your previous sync executor
    result = execute_sync(action, args, meta)

    # 2) If the result is a coroutine/future/task → await
    if inspect.isawaitable(result):
        job = recorder.job_start(tool_call_ctx, meta)
        try:
            value = await result
            recorder.job_finish(job.job_id, ok=True, result=value if isinstance(value, dict) else {"result": value})
            return {"job": job.model_dump(), "result": value}
        except Exception as e:
            recorder.job_finish(job.job_id, ok=False, error=str(e))
            raise

    # 3) If the result is an async stream → stream + persist chunks, return StreamRef
    stream = await _maybe_stream(result)
    if stream is not None:
        sem = describe_type(meta.get("return_type")) if meta.get("return_type") else None
        item_schema = sem.item_schema if sem and sem.item_schema else None
        sref = recorder.stream_start(tool_call_ctx, item_schema=item_schema, transport="ndjson")
        seq = 0
        async for item in stream:
            data = item.model_dump() if hasattr(item, "model_dump") else item
            recorder.stream_append(sref.stream_id, seq, data)
            seq += 1
        return {"stream": sref.model_dump(), "chunks": seq}

    # 4) Immediate result
    return {"result": result}

    The gateway can decide: return the immediate result, or if you prefer non‑blocking, always return JobRef and let clients poll/subscribe.

TriggerEvent = Literal["before_call", "after_call", "on_error", "on_chunk", "finally"]

    sample_rate: float = 1.0                   # 0..1 sampling
    predicate: Optional[str] = None            # mini-expr, see AssistEngine._predicate_ok

# ── Selection ─────────────────────────────────────────────────────────────────

class SelectorSpec(BaseModel):

    JSON-ish path selector. Minimal subset: "$.a.b[*].c" | "results.*.url"
    """
    path: str

# ── Transforms ────────────────────────────────────────────────────────────────

class CanonicalizeURL(BaseModel):
    drop_fragments: bool = True
    strip_query_params: List[str] = Field(default_factory=lambda: ["utm_", "gclid"])
    sort_query: bool = True
    force_lower_host: bool = True
    drop_default_ports: bool = True

    def __call__(self, url: str) -> str:
        try:
            u = urlparse(url)
            scheme = u.scheme
            netloc = u.hostname.lower() if (self.force_lower_host and u.hostname) else (u.hostname or "")
            if u.port and not (self.drop_default_ports and ((scheme == "http" and u.port == 80) or (scheme == "https" and u.port == 443))):
                netloc = f"{netloc}:{u.port}"
            path = u.path or "/"
            # query
            pairs = parse_qsl(u.query, keep_blank_values=False)
            filtered = [(k, v) for k, v in pairs if not any(k.startswith(prefix) for prefix in self.strip_query_params)]
            if self.sort_query:
                filtered.sort()
            query = urlencode(filtered)
            fragment = "" if self.drop_fragments else u.fragment
            return urlunparse((scheme, netloc, path, "", query, fragment))
        except Exception:
            return url

class FilterByDomain(BaseModel):
    allowlist: List[str] = Field(default_factory=list)   # supports "*.example.com"
    denylist: List[str] = Field(default_factory=list)

    def _match(self, host: str, pattern: str) -> bool:
        if pattern.startswith("*."):
            return host == pattern[2:] or host.endswith(pattern[1:])
        return host == pattern

    def __call__(self, url: str) -> bool:
        try:
            host = urlparse(url).hostname or ""
        except Exception:
            return False
        if self.denylist and any(self._match(host, p) for p in self.denylist):
            return False
        if self.allowlist:
            return any(self._match(host, p) for p in self.allowlist)
        return True

class TransformChain(BaseModel):
    canonicalize: Optional[CanonicalizeURL] = CanonicalizeURL()
    domain_filter: Optional[FilterByDomain] = None
    unique: bool = True

    def run(self, urls: List[str]) -> List[str]:
        out: List[str] = []
        seen: set[str] = set()
        for u in urls:
            c = self.canonicalize(u) if self.canonicalize else u
            if self.domain_filter and not self.domain_filter(c):
                continue
            if self.unique:
                if c in seen:
                    continue
                seen.add(c)
            out.append(c)
        return out

# ── tiny path getter ("$.results[*].url" and "results.*.url") ─────────────────
def _get_all(root: Any, path: str) -> List[Any]:
    # minimal: split by dots, support "$." prefix and "*" wildcard on lists/dicts
    if path.startswith("$."): path = path[2:]
    parts = [p for p in path.split(".") if p]
    curr: List[Any] = [root]
    for p in parts:
        nxt: List[Any] = []
        is_star = p in ("*", "[*]")
        for obj in curr:
            if isinstance(obj, dict):
                if is_star:
                    nxt.extend(obj.values())
                elif p in obj:
                    nxt.append(obj[p])
            elif isinstance(obj, list):
                if is_star or p in ("[*]", "*"):
                    nxt.extend(obj)
                else:
                    try:
                        i = int(p)
                        if 0 <= i < len(obj):
                            nxt.append(obj[i])
                    except Exception:
                        pass
        curr = nxt
    # flatten nested lists of scalars
    out: List[Any] = []
    for x in curr:
        if isinstance(x, list): out.extend(x)
        else: out.append(x)
    return out

# ── simple predicate ──────────────────────────────────────────────────────────
def _predicate_ok(expr: Optional[str], ctx: Dict[str, Any]) -> bool:
    if not expr: return True
    # safe eval against a tiny context
    SAFE = {"len": len, "any": any, "all": all}
    try:
        return bool(eval(expr, {"__builtins__": {}}, {**SAFE, **ctx}))
    except Exception:
        return False

# ── Seen store (memory stub; swap with Redis/HTTP) ────────────────────────────
class SeenStore:
    def __init__(self): self._m = {}
    @staticmethod
    def _key(ns: str, url: str) -> str: return f"{ns}:{hashlib_sha(url)}"
    def has(self, ns: str, url: str) -> bool: return self._key(ns,url) in self._m
    def put(self, ns: str, url: str, ttl: int): self._m[self._key(ns,url)] = (time.time()+ttl)
    def expire(self):
        now=time.time()
        for k,(till) in list(self._m.items()):
            if till < now: del self._m[k]

def hashlib_sha(s: str) -> str:
    import hashlib; return hashlib.sha256(s.encode()).hexdigest()

GLOBAL_SEEN = SeenStore()

# ── HTTP enqueue sink (stub) ──────────────────────────────────────────────────
async def _enqueue_http(endpoint: str, items: List[Dict[str, Any]], auth_header: Optional[str] = None):
    if not endpoint or not items: return
    headers = {"Content-Type": "application/json"}
    if auth_header: headers["Authorization"] = auth_header
    async with httpx.AsyncClient(timeout=10.0) as c:
        await c.post(endpoint, json={"items": items}, headers=headers)

# ── Prefetch (best-effort) ────────────────────────────────────────────────────
async def _prefetch(urls: List[str], method: str = "HEAD", timeout_ms: int = 4000, max_concurrency: int = 16):
    if not urls: return
    sem = asyncio.Semaphore(max_concurrency)
    async def one(u: str):
        async with sem:
            try:
                async with httpx.AsyncClient(timeout=timeout_ms/1000.0) as c:
                    if method == "HEAD": await c.head(u)
                    else: await c.get(u)
            except Exception: pass
    await asyncio.gather(*(one(u) for u in urls))

# ── Graph write (URL facts) ───────────────────────────────────────────────────
async def _graph_write(service_id: str, urls: List[str], meta: Dict[str, Any], url_label="URL", rel_type="DISCOVERED"):
    if not urls or not service_id: return
    gc = get_graph_client(service_id)
    for u in urls:
        key = {"url": u}
        props = {k: v for k,v in meta.items() if v is not None}
        gc.merge_node(url_label, key, props)
        if "source" in props:
            # Optional: link (:Source {id}) or (:Query {q})
            pass
        # You can enrich with domain:
        host = urlparse(u).hostname or ""
        gc.merge_node("Domain", {"name": host}, {})
        gc.merge_rel(url_label, key, "HOSTED_ON", "Domain", {"name": host}, {})
        # If you have a current Query/context node id:
        if "query_node_id" in props:
            gc.merge_rel("Query", {"id": props["query_node_id"]}, rel_type, url_label, key, {"ts": time.time()})

# ── AssistEngine ──────────────────────────────────────────────────────────────
class AssistEngine:
    def __init__(self, spec: AssistSpec):
        self.spec = spec

    def _extract_urls(self, payload: Any, sel: str) -> List[str]:
        vals = _get_all(payload, sel)
        flat: List[str] = []
        for v in vals:
            if isinstance(v, str): flat.append(v)
            elif isinstance(v, dict):
                # if dict has 'url' field
                if "url" in v and isinstance(v["url"], str):
                    flat.append(v["url"])
        return flat

    async def run(self, *, event: str, payload: Any, context: Dict[str, Any]) -> Dict[str, Any]:
        """
        event: "before_call" | "after_call" | "on_error" | "on_chunk" | "finally"
        payload: args (before), result (after), chunk (on_chunk)
        context: {tool_name, action_name, service_id, conversation_id, query, ...}
        """
        if not self.spec.active: return {}
        results: Dict[str, Any] = {}
        for op in self.spec.ops:
            if op.trigger.event != event: continue
            if op.trigger.sample_rate < 1.0:
                import random
                if random.random() > op.trigger.sample_rate:
                    continue
            if not _predicate_ok(op.trigger.predicate, {"payload": payload, "ctx": context}):
                continue

            # 1) select
            urls = self._extract_urls(payload, op.selector.path)
            if not urls: continue

            # 2) transform
            urls = op.transforms.run(urls)

            # 3) seen/dedupe
            new_urls: List[str] = []
            if op.seen_store:
                ns = op.seen_store.namespace
                for u in urls:
                    if not GLOBAL_SEEN.has(ns, u):
                        new_urls.append(u)
                        GLOBAL_SEEN.put(ns, u, op.seen_store.ttl_seconds)
            else:
                new_urls = urls

            meta = {
                "ts": time.time(),
                "source": f"{context.get('tool_name')}.{context.get('action_name')}",
                **(op.meta or {}),
                **{k: context.get(k) for k in ("query","conversation_id","user_id")}
            }

            # 4) enqueue
            if op.enqueue and new_urls:
                if op.enqueue.type == "http" and op.enqueue.endpoint:
                    batch = op.enqueue.batch or 50
                    chunks = [new_urls[i:i+batch] for i in range(0, len(new_urls), batch)]
                    for ch in chunks:
                        await _enqueue_http(op.enqueue.endpoint, [{"url": u, **meta} for u in ch], auth_header=op.enqueue.auth_header)
                # other transports (kafka/sqs/redis) can be plugged here

            # 5) prefetch
            if op.prefetch and new_urls:
                await _prefetch(new_urls, method=op.prefetch.method, timeout_ms=op.prefetch.timeout_ms, max_concurrency=op.prefetch.max_concurrency)

            # 6) graph writes
            if op.graph and (urls or new_urls):
                try:
                    await _graph_write(op.graph.service_id, new_urls, meta, url_label=op.graph.url_label, rel_type=op.graph.rel_type)
                except Exception:
                    pass

            # 7) metrics (placeholder; wire to Prometheus or your metrics bus)
            if op.metrics:
                results.setdefault("metrics", []).append({
                    "namespace": op.metrics.namespace,
                    "urls_discovered": len(urls),
                    "urls_new": len(new_urls),
                })

            # collect per-op outcome
            results.setdefault("ops", {})[op.name] = {"discovered": len(urls), "new": len(new_urls)}
        return results

def execute(action: ToolAction, args: Dict[str, Any], meta: Dict[str, Any]) -> Dict[str, Any]:
    ctx = {
        "tool_name": meta.get("tool_name") or "",
        "action_name": action.name,

        **meta,

        # fire and forget; sidecars must never block primary call outcome

    # after_call

        ae = AssistEngine(action.assist)
        try:
            asyncio.run(ae.run(event="after_call", payload=out, context=ctx))
        except Exception: pass

    return out

6) Why this works well for your stack

    Declarative: the sidecar logic is data‑driven (AssistSpec), versionable, and can be toggled per action/tool/toolkit/service.

    Non‑intrusive: AssistEngine runs outside the main success path; failures never break the user‑visible result.

    Composable: chain transforms, plug different sinks (HTTP/Kafka/SQS), and reuse selectors across actions.

    Graph‑aware: optional write‑through to your graph overlay so retrieval can reason over URL nodes, domains, and provenance.

    Stream‑ready: the same engine can fire on on_chunk for streaming search providers (paginate, incremental SERP).

    Idempotent: SeenStore + canonicalization keeps enqueue clean and bounded.

7) Production knobs you’ll likely add

    Policy: allow/deny sinks per environment; redact PII in sidecar metadata.

    Backpressure: max total sidecar concurrency per action; circuit‑breaker when queue is throttled.

    Telem: emit traces with correlation_id (you already have it on ToolExecutionContext).

    Selectors: add JMESPath/JSONPath proper; current selector is a tiny subset to stay dependency‑free.

    Bloom/HLL: replace memory seen store with Redis Bloom or a KV service for cross‑process dedupe.

    Retries: exponential backoff for enqueue/prefetch; dead‑letter queues.

    Per‑op metrics: timings, errors, sampled payloads.
